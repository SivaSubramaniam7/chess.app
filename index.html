import javax.swing.*; import java.awt.*; import java.awt.event.*; import java.util.ArrayList; import java.util.List; /** * A simple two-player chess app using Java Swing. * Uppercase letters = white pieces, lowercase = black pieces, ' ' = empty. * Row 0 is the top of the screen (black side), row 7 is the bottom (white side). * * Supported: all normal piece moves, turn order, check, checkmate, stalemate, * pawn promotion (automatically to a queen). * Not included: castling, en passant, draw by repetition. * * Run: javac ChessApp.java then java ChessApp */ public class ChessApp extends JPanel { static final int SQ = 80; // size of one square in pixels char[][] board = new char[8][8]; boolean whiteTurn = true; boolean gameOver = false; int selRow = -1, selCol = -1; // currently selected square List legalForSelected = new ArrayList<>(); String status = "White to move"; public ChessApp() { setPreferredSize(new Dimension(8 * SQ, 8 * SQ + 40)); reset(); addMouseListener(new MouseAdapter() { @Override public void mousePressed(MouseEvent e) { handleClick(e.getY() / SQ, e.getX() / SQ); } }); } // ---------- Game setup ---------- void reset() { String back = "rnbqkbnr"; for (int r = 0; r < 8; r++) for (int c = 0; c < 8; c++) board[r][c] = ' '; for (int c = 0; c < 8; c++) { board[0][c] = back.charAt(c); board[1][c] = 'p'; board[6][c] = 'P'; board[7][c] = Character.toUpperCase(back.charAt(c)); } whiteTurn = true; gameOver = false; selRow = selCol = -1; legalForSelected.clear(); status = "White to move"; repaint(); } // ---------- Input handling ---------- void handleClick(int r, int c) { if (gameOver || r < 0 || r > 7 || c < 0 || c > 7) return; // 1) Clicked a highlighted destination -> make the move for (int[] m : legalForSelected) { if (m[0] == r && m[1] == c) { makeMove(selRow, selCol, r, c); return; } } // 2) Clicked one of your own pieces -> select it char p = board[r][c]; if (p != ' ' && isWhite(p) == whiteTurn) { selRow = r; selCol = c; legalForSelected = legalMoves(r, c); } else { // 3) Clicked elsewhere -> deselect selRow = selCol = -1; legalForSelected.clear(); } repaint(); } void makeMove(int fr, int fc, int tr, int tc) { char piece = board[fr][fc]; board[tr][tc] = piece; board[fr][fc] = ' '; // Pawn promotion if (piece == 'P' && tr == 0) board[tr][tc] = 'Q'; if (piece == 'p' && tr == 7) board[tr][tc] = 'q'; whiteTurn = !whiteTurn; selRow = selCol = -1; legalForSelected.clear(); updateStatus(); repaint(); } void updateStatus() { String side = whiteTurn ? "White" : "Black"; String other = whiteTurn ? "Black" : "White"; boolean check = inCheck(whiteTurn); if (!hasAnyLegalMove(whiteTurn)) { gameOver = true; status = check ? "Checkmate! " + other + " wins." : "Stalemate - draw."; } else { status = side + " to move" + (check ? " - CHECK!" : ""); } } // ---------- Chess rules ---------- static boolean isWhite(char p) { return Character.isUpperCase(p); } static boolean inside(int r, int c) { return r >= 0 && r < 8 && c >= 0 && c < 8; } /** Adds (r,c) if it is on the board and empty or holds an enemy piece. */ void addIfOk(List list, int r, int c, boolean white) { if (!inside(r, c)) return; if (board[r][c] == ' ' || isWhite(board[r][c]) != white) list.add(new int[]{r, c}); } /** Slide in a direction until blocked (used by bishop, rook, queen). */ void slide(List list, int r, int c, int dr, int dc, boolean white) { int nr = r + dr, nc = c + dc; while (inside(nr, nc)) { if (board[nr][nc] == ' ') { list.add(new int[]{nr, nc}); } else { if (isWhite(board[nr][nc]) != white) list.add(new int[]{nr, nc}); break; } nr += dr; nc += dc; } } /** * Moves a piece could make ignoring whether they leave its own king in check. * If attackOnly is true, pawns only report their diagonal attack squares. */ List pieceMoves(int r, int c, boolean attackOnly) { List list = new ArrayList<>(); char p = board[r][c]; boolean white = isWhite(p); switch (Character.toLowerCase(p)) { case 'p': { int dir = white ? -1 : 1; int startRow = white ? 6 : 1; if (!attackOnly) { if (inside(r + dir, c) && board[r + dir][c] == ' ') { list.add(new int[]{r + dir, c}); if (r == startRow && board[r + 2 * dir][c] == ' ') list.add(new int[]{r + 2 * dir, c}); } } for (int dc = -1; dc <= 1; dc += 2) { int nr = r + dir, nc = c + dc; if (!inside(nr, nc)) continue; if (attackOnly) list.add(new int[]{nr, nc}); else if (board[nr][nc] != ' ' && isWhite(board[nr][nc]) != white) list.add(new int[]{nr, nc}); } break; } case 'n': { int[][] jumps = {{-2, -1}, {-2, 1}, {-1, -2}, {-1, 2}, {1, -2}, {1, 2}, {2, -1}, {2, 1}}; for (int[] j : jumps) addIfOk(list, r + j[0], c + j[1], white); break; } case 'k': { for (int dr = -1; dr <= 1; dr++) for (int dc = -1; dc <= 1; dc++) if (dr != 0 || dc != 0) addIfOk(list, r + dr, c + dc, white); break; } case 'b': slide(list, r, c, -1, -1, white); slide(list, r, c, -1, 1, white); slide(list, r, c, 1, -1, white); slide(list, r, c, 1, 1, white); break; case 'r': slide(list, r, c, -1, 0, white); slide(list, r, c, 1, 0, white); slide(list, r, c, 0, -1, white); slide(list, r, c, 0, 1, white); break; case 'q': for (int dr = -1; dr <= 1; dr++) for (int dc = -1; dc <= 1; dc++) if (dr != 0 || dc != 0) slide(list, r, c, dr, dc, white); break; } return list; } /** Is the given side's king currently under attack? */ boolean inCheck(boolean white) { char king = white ? 'K' : 'k'; int kr = -1, kc = -1; for (int r = 0; r < 8; r++) for (int c = 0; c < 8; c++) if (board[r][c] == king) { kr = r; kc = c; } if (kr == -1) return false; for (int r = 0; r < 8; r++) { for (int c = 0; c < 8; c++) { char p = board[r][c]; if (p == ' ' || isWhite(p) == white) continue; // only enemy pieces for (int[] m : pieceMoves(r, c, true)) if (m[0] == kr && m[1] == kc) return true; } } return false; } /** Moves that do not leave your own king in check. */ List legalMoves(int r, int c) { List result = new ArrayList<>(); boolean white = isWhite(board[r][c]); for (int[] m : pieceMoves(r, c, false)) { // Try the move, test for check, then undo it char moving = board[r][c]; char captured = board[m[0]][m[1]]; board[m[0]][m[1]] = moving; board[r][c] = ' '; boolean safe = !inCheck(white); board[r][c] = moving; board[m[0]][m[1]] = captured; if (safe) result.add(m); } return result; } boolean hasAnyLegalMove(boolean white) { for (int r = 0; r < 8; r++) for (int c = 0; c < 8; c++) if (board[r][c] != ' ' && isWhite(board[r][c]) == white && !legalMoves(r, c).isEmpty()) return true; return false; } // ---------- Drawing ---------- static String symbol(char p) { switch (p) { case 'K': return "\u2654"; case 'Q': return "\u2655"; case 'R': return "\u2656"; case 'B': return "\u2657"; case 'N': return "\u2658"; case 'P': return "\u2659"; case 'k': return "\u265A"; case 'q': return "\u265B"; case 'r': return "\u265C"; case 'b': return "\u265D"; case 'n': return "\u265E"; case 'p': return "\u265F"; default: return ""; } } @Override protected void paintComponent(Graphics g0) { super.paintComponent(g0); Graphics2D g = (Graphics2D) g0; g.setRenderingHint(RenderingHints.KEY_ANTIALIASING, RenderingHints.VALUE_ANTIALIAS_ON); g.setRenderingHint(RenderingHints.KEY_TEXT_ANTIALIASING, RenderingHints.VALUE_TEXT_ANTIALIAS_ON); Font pieceFont = new Font("Dialog", Font.PLAIN, 54); g.setFont(pieceFont); FontMetrics fm = g.getFontMetrics(); for (int r = 0; r < 8; r++) { for (int c = 0; c < 8; c++) { // Square colour boolean light = (r + c) % 2 == 0; g.setColor(light ? new Color(240, 217, 181) : new Color(181, 136, 99)); if (r == selRow && c == selCol) g.setColor(new Color(246, 246, 105)); g.fillRect(c * SQ, r * SQ, SQ, SQ); // Piece char p = board[r][c]; if (p != ' ') { String s = symbol(p); int x = c * SQ + (SQ - fm.stringWidth(s)) / 2; int y = r * SQ + (SQ + fm.getAscent() - fm.getDescent()) / 2; g.setColor(Color.BLACK); g.drawString(s, x, y); } } } // Legal move markers g.setColor(new Color(0, 130, 0, 150)); for (int[] m : legalForSelected) { int cx = m[1] * SQ + SQ / 2, cy = m[0] * SQ + SQ / 2; if (board[m[0]][m[1]] == ' ') g.fillOval(cx - 10, cy - 10, 20, 20); else { g.setStroke(new BasicStroke(4)); g.drawOval(cx - 34, cy - 34, 68, 68); } } // Status bar g.setColor(new Color(40, 40, 40)); g.fillRect(0, 8 * SQ, 8 * SQ, 40); g.setColor(Color.WHITE); g.setFont(new Font("SansSerif", Font.BOLD, 18)); g.drawString(status, 12, 8 * SQ + 26); } // ---------- Main ---------- public static void main(String[] args) { SwingUtilities.invokeLater(() -> { JFrame frame = new JFrame("Chess"); ChessApp game = new ChessApp(); JButton newGame = new JButton("New Game"); newGame.addActionListener(e -> game.reset()); frame.setLayout(new BorderLayout()); frame.add(game, BorderLayout.CENTER); frame.add(newGame, BorderLayout.SOUTH); frame.pack(); frame.setResizable(false); frame.setLocationRelativeTo(null); frame.setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE); frame.setVisible(true); }); } }
